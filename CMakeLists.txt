cmake_minimum_required(VERSION 3.22)
project(LonglandFlock VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_DIR "C:/SDK/JUCE" CACHE PATH "JUCE source directory")
add_subdirectory("${JUCE_DIR}" JUCE)
set(formats VST3)
if(APPLE)
  list(APPEND formats AU)
endif()
juce_add_plugin(LonglandFlock COMPANY_NAME Longland PLUGIN_MANUFACTURER_CODE Lngl PLUGIN_CODE Lflk FORMATS ${formats} PRODUCT_NAME "Longland Flock" NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_SYNTH FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(LonglandFlock PRIVATE Source/Plugin.cpp)
target_compile_definitions(LonglandFlock PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(LonglandFlock PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
option(FLOCK_BUILD_TESTS "Build developer-only verification tools" OFF)
if(FLOCK_BUILD_TESTS)
add_executable(FlockTests Source/Tests.cpp)
enable_testing()
add_test(NAME FlockTests COMMAND FlockTests)
add_executable(FlockMeasure Source/Measure.cpp)
add_executable(DelayTests Source/DelayTests.cpp)
add_executable(BrainTests Source/BrainTests.cpp)
add_test(NAME BrainTests COMMAND BrainTests)
add_test(NAME DelayTests COMMAND DelayTests)
add_test(NAME FlockMeasure COMMAND FlockMeasure)
juce_add_console_app(FlockIntegration PRODUCT_NAME "Flock Integration Tests")
target_sources(FlockIntegration PRIVATE Source/Integration.cpp)
target_compile_definitions(FlockIntegration PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_STANDALONE_APPLICATION=1 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(FlockIntegration PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
add_test(NAME FlockIntegration COMMAND FlockIntegration)
endif()
